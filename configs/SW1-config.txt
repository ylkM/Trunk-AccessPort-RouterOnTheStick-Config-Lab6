! ============================================
! SW1 Configuration
! Access ports for VLAN10 + VLAN30, trunk to SW2
! ============================================

enable
configure terminal

hostname SW1

! --- Create VLANs (all 3 + unused native, even though SW1 has no VLAN20 ports --
! required so the trunk can carry VLAN20 through to SW2/R1 and for STP consistency) ---
vlan 10
 name VLAN10
 exit

vlan 20
 name VLAN20
 exit

vlan 30
 name VLAN30
 exit

vlan 999
 name NATIVE_UNUSED
 exit

! --- Access ports: VLAN10 PCs ---
interface FastEthernet0/1
 description PC1 (10.0.0.1)
 switchport mode access
 switchport access vlan 10
 exit

interface FastEthernet0/2
 description PC2 (10.0.0.2)
 switchport mode access
 switchport access vlan 10
 exit

! --- Access ports: VLAN30 PCs ---
interface FastEthernet0/3
 description PC6 (10.0.0.129)
 switchport mode access
 switchport access vlan 30
 exit

interface FastEthernet0/4
 description PC7 (10.0.0.130)
 switchport mode access
 switchport access vlan 30
 exit

! --- Trunk to SW2 ---
interface GigabitEthernet0/1
 description Trunk to SW2 Gig0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30
 exit

end
write memory

! ============================================
! Verification commands
! ============================================
! show vlan brief
! show interfaces trunk
